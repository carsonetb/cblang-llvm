; ModuleID = "test"
target triple = "unknown-unknown-unknown"
target datalayout = ""

%"TestClass" = type {i32, i32 ()*}
declare i8* @"malloc"(i32 %".1")

declare i8* @"realloc"(i8* %".1", i32 %".2")

declare void @"free"(i8* %".1")

declare i32 @"printf"(i8* %".1", ...)

define i8* @"rc_alloc"(i32 %".1")
{
entry:
  %"total_size" = add i32 %".1", 4
  %"mem" = call i8* @"malloc"(i32 %"total_size")
  %"count_ptr" = bitcast i8* %"mem" to i32*
  store i32 1, i32* %"count_ptr"
  %"data" = getelementptr i8, i8* %"mem", i32 4
  ret i8* %"data"
}

define void @"rc_retain"(i8* %".1")
{
entry:
  %"header" = getelementptr i8, i8* %".1", i32 -4
  %"count_ptr" = bitcast i8* %"header" to i32*
  %"old_count" = load i32, i32* %"count_ptr"
  %"new_count" = add i32 %"old_count", 1
  store i32 %"new_count", i32* %"count_ptr"
  ret void
}

define void @"rc_release"(i8* %".1", void (i8*)* %".2")
{
entry:
  %"header_ptr" = getelementptr i8, i8* %".1", i32 -4
  %"count_ptr" = bitcast i8* %"header_ptr" to i32*
  %"old_count" = load i32, i32* %"count_ptr"
  %"new_count" = sub i32 %"old_count", 1
  store i32 %"new_count", i32* %"count_ptr"
  %"is_zero" = icmp eq i32 %"new_count", 0
  br i1 %"is_zero", label %"free_block", label %"end"
free_block:
  %"has_dtor" = icmp ne void (i8*)* %".2", null
  br i1 %"has_dtor", label %"free_call_dtor", label %"free_do_free"
free_call_dtor:
  call void %".2"(i8* %".1")
  br label %"free_do_free"
free_do_free:
  call void @"free"(i8* %"header_ptr")
  br label %"end"
end:
  ret void
}

define void @"string_destructor"(i8* %".1")
{
entry:
  %"start_ptr_ptr" = bitcast i8* %".1" to i8**
  %"start_ptr" = load i8*, i8** %"start_ptr_ptr"
  call void @"free"(i8* %"start_ptr")
  ret void
}

define i32 @"main"()
{
entry:
  call void @"main__test__main"()
  ret i32 0
}

define void @"TestClass_destructor"(i8* %".1")
{
entry:
  %"self_ptr" = bitcast i8* %".1" to %"TestClass"*
  ret void
}

define %"TestClass"* @"main__test__TestClass__init"()
{
entry:
  %"size_ptr" = getelementptr %"TestClass", %"TestClass"* null, i32 1
  %"size" = ptrtoint %"TestClass"* %"size_ptr" to i32
  %"raw_ptr" = call i8* @"rc_alloc"(i32 %"size")
  %"this_ptr" = bitcast i8* %"raw_ptr" to %"TestClass"*
  %"value_ptr" = alloca i32
  store i32 0, i32* %"value_ptr"
  %"load_value" = load i32, i32* %"value_ptr"
  %"loaded" = getelementptr %"TestClass", %"TestClass"* %"this_ptr", i32 0, i32 0
  store i32 %"load_value", i32* %"loaded"
  ret %"TestClass"* %"this_ptr"
}

define i32 @"main__test__TestClass__num2"(%"TestClass"* %".1")
{
entry:
  %"x" = getelementptr %"TestClass", %"TestClass"* %".1", i32 0, i32 0
  %"num2" = getelementptr %"TestClass", %"TestClass"* %".1", i32 0, i32 1
  %"load_value" = load i32, i32* %"x"
  ret i32 %"load_value"
}

define i32 @"main__test__num"()
{
entry:
  %"TestClass" = call %"TestClass"* @"main__test__TestClass__init"()
  %"internal_val_mem" = getelementptr %"TestClass", %"TestClass"* %"TestClass", i32 0, i32 0
  %"internal_val" = load i32, i32* %"internal_val_mem"
  %"value_ptr" = alloca i32
  store i32 %"internal_val", i32* %"value_ptr"
  %"raw_ptr" = bitcast %"TestClass"* %"TestClass" to i8*
  call void @"rc_release"(i8* %"raw_ptr", void (i8*)* @"TestClass_destructor")
  %"load_value" = load i32, i32* %"value_ptr"
  ret i32 %"load_value"
}

define void @"main__test__main"()
{
entry:
  %"mem" = call i8* @"malloc"(i32 13)
  %"alloced_array" = bitcast i8* %"mem" to [13 x i8]*
  store [13 x i8] c"Hello World!\00", [13 x i8]* %"alloced_array"
  %"value_memory" = call i8* @"rc_alloc"(i32 8)
  %"value_ptr" = bitcast i8* %"value_memory" to i8**
  store i8* %"mem", i8** %"value_ptr"
  %"raw_ptr" = bitcast i8** %"value_ptr" to i8*
  call void @"rc_retain"(i8* %"raw_ptr")
  %"load_value" = load i8*, i8** %"value_ptr"
  %"printf" = call i32 (i8*, ...) @"printf"(i8* %"load_value")
  %"value_ptr.1" = alloca i32
  store i32 %"printf", i32* %"value_ptr.1"
  %"num" = call i32 @"main__test__num"()
  %"value_ptr.2" = alloca i32
  store i32 %"num", i32* %"value_ptr.2"
  %"raw_ptr.1" = bitcast i8** %"value_ptr" to i8*
  call void @"rc_release"(i8* %"raw_ptr.1", void (i8*)* @"string_destructor")
  ret void
}
